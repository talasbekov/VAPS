# Правка `plan` в `plane_task.py` (26.08.2026, Plane №140)

Скрипт лежит в `/home/erda/plane/migration/plane_task.py` — это НЕ репозиторий проекта,
поэтому правка нигде не версионируется. Копия тела команды `plan` хранится здесь: пропадёт
машина или кто-то откатит файл — восстанавливать отсюда.

Суть: задачами становятся ТОЛЬКО строки списка внутри раздела «Шаги». До этого разбирался
весь файл, и в трекер уезжали разведка, слова заказчика и отвергнутые варианты.

```python
        # Шаги плана → задачи. Шагом считается строка списка верхнего уровня
        # («- », «* », «1. ») ВНУТРИ РАЗДЕЛА «Шаги» — вложенные строки уезжают в
        # описание своего шага, чтобы подпункты не расползлись в отдельные
        # карточки.
        #
        # 🔴 РАЗДЕЛ ОБЯЗАТЕЛЕН (Plane №140). Раньше разбирался ВЕСЬ файл, и
        # задачами становились строки списков из «Что уже есть», «Решения
        # заказчика», «Отвергнутые варианты» и «Риски» — то есть в трекер
        # уезжали факты разведки и отвергнутые идеи, помеченные как работа. В
        # плане ролей и прав это дало бы 20 карточек вместо 10, причём половина
        # описывала бы то, чего делать НЕ надо.
        dry_run = "--dry-run" in sys.argv[4:]
        source = Path(sys.argv[3]).read_text(encoding="utf-8").splitlines()
        steps, current, inside = [], None, False
        for raw in source:
            heading = re.match(r"^(#{1,6})\s+(.+?)\s*$", raw)
            if heading:
                # Заголовок закрывает раздел шагов и открывает его, если это
                # он сам. Вложенные заголовки внутри «Шагов» тоже закрывают:
                # шаг — строка списка, а не подраздел.
                inside = heading.group(2).strip().lower().startswith("шаг")
                current = None
                continue
            if not inside:
                continue
            top = re.match(r"^\s{0,1}(?:[-*]|\d+[.)])\s+(.+)$", raw)
            if top:
                current = {"name": top.group(1).strip(), "body": []}
                steps.append(current)
            elif current is not None and raw.strip():
                current["body"].append(raw.strip())
        if not steps:
            raise SystemExit(
                "в файле нет раздела «Шаги» со строками списка. Задачами "
                "становятся ТОЛЬКО шаги: заголовок вида «## Шаги», под ним "
                "строки «- ...». Так в трекер не уезжают разведка, решения и "
                "отвергнутые варианты (Plane №140)."
            )
        if dry_run:
            print(f"РАЗБОР (задачи НЕ заводятся), шагов: {len(steps)}")
            for index, step in enumerate(steps, start=1):
                print(f"  {index}. {step['name']}")
                for line in step["body"]:
                    print(f"       {line[:100]}")
            return
        for step in steps:
            print(_create(pid, st, step["name"], "\n".join(step["body"]), "Backlog"))
        print(f"задач заведено: {len(steps)}")
        return
```
